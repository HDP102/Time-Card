$pce = "https://us-scp41.illum.io:443"
$org = "4325389"
[Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12

$id  = Read-Host "API key ID"
$sec = Read-Host "API secret" -AsSecureString
$plain = [Runtime.InteropServices.Marshal]::PtrToStringAuto([Runtime.InteropServices.Marshal]::SecureStringToBSTR($sec))
$h = @{ Authorization = "Basic " + [Convert]::ToBase64String([Text.Encoding]::ASCII.GetBytes("$($id):$plain")) }
Remove-Variable plain

# Is hit count enabled on the PCE?
Invoke-RestMethod -Uri "$pce/api/v2/orgs/$org/report_templates/rule_hit_count_report" -Headers $h

# Which VEN scopes have it enabled?
(Invoke-RestMethod -Uri "$pce/api/v2/orgs/$org/sec_policy/active/firewall_settings" -Headers $h).rule_hit_count_enabled_scopes
