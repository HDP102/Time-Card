# 1. Find OpenSSL inside your Git install
$o = Get-ChildItem "$env:LOCALAPPDATA\Programs\Git" -Recurse -Filter openssl.exe -ErrorAction SilentlyContinue | Select-Object -First 1 -ExpandProperty FullName
Set-Alias openssl $o
openssl version

# 2. Folder outside the repo
mkdir C:\Users\H2PR\certs\mea-lab -Force
cd C:\Users\H2PR\certs\mea-lab

# 3. Private key + self-signed certificate (2 years)
openssl req -x509 -newkey rsa:2048 -sha256 -days 730 -nodes -keyout nps_fw_mea_lab.key -out nps_fw_mea_lab.crt -subj "/CN=NPS-Firewall-Automation-MEA-Lab/O=Pacific Gas and Electric Company"

# 4. SHA-1 thumbprint (keep this)
openssl x509 -in nps_fw_mea_lab.crt -noout -fingerprint -sha1

# 5. Check the certificate details
openssl x509 -in nps_fw_mea_lab.crt -noout -subject -dates

# 6. Lock the private key to your account only
icacls nps_fw_mea_lab.key /inheritance:r /grant:r "${env:USERNAME}:(R)"

# 7. Confirm both files exist
dir
